<?php
declare(strict_types=1);

namespace App\Repository;

use App\Database\DB;

final class HistoryRepository
{
    public function __construct(private readonly DB $db)
    {
    }

    public function listByCall(int $callId): array
    {
        return $this->db->fetchAll(
            "SELECT * FROM lift_history WHERE call_id = :id ORDER BY history_date ASC",
            ['id' => $callId]
        );
    }

    /**
     * Запись действия в историю заявки с указанием источника
     * @param string $source Источник действия ('online' | 'offline_sync' | 'spult')
     */
    public function add(int $callId, string $info, ?int $userId = null, string $source = 'online'): void
    {
        // Если в вашей таблице lift_history пока нет колонки source,
        // текст источника склеивается с информацией, чтобы не ломать структуру до миграции
        $historyText = $source === 'offline_sync' ? "[Офлайн] {$info}" : $info;

        $this->db->execute(
            "INSERT INTO lift_history (history_date, history_info, call_id) VALUES (?, ?, ?)",
            [time(), $historyText, $callId]
        );
    }
}
    /**
     * Поиск активного дубликата заявки по критическим параметрам
     */
    public function findActiveDuplicate(int $liftId, int $groupId, int $departmentId): ?array
    {
        $sql = "SELECT call_id, call_details, call_status
                FROM lift_calls
                WHERE address_lift = :lift_id
                  AND call_group = :group_id
                  AND call_department = :department_id
                  AND call_status = 0
                LIMIT 1";

        return $this->db->fetchOne($sql, [
            'lift_id'       => $liftId,
            'group_id'      => $groupId,
            'department_id' => $departmentId
        ]);
    }
