DELIMITER $$

DROP PROCEDURE IF EXISTS `add_column_if_not_exists`$$
CREATE PROCEDURE `add_column_if_not_exists`(
    IN table_name VARCHAR(64),
    IN column_name VARCHAR(64),
    IN column_definition VARCHAR(500)
)
BEGIN
    SET @dbname = DATABASE();
    SET @exists = (
        SELECT COUNT(*) 
        FROM information_schema.COLUMNS 
        WHERE TABLE_SCHEMA = @dbname 
          AND TABLE_NAME = table_name 
          AND COLUMN_NAME = column_name
    );
    
    IF @exists = 0 THEN
        SET @sql = CONCAT('ALTER TABLE `', table_name, '` ADD COLUMN `', column_name, '` ', column_definition);
        PREPARE stmt FROM @sql;
        EXECUTE stmt;
        DEALLOCATE PREPARE stmt;
    END IF;
END$$

    IN idx_definition VARCHAR(500)
)
BEGIN
    SET @dbname = DATABASE();
    SET @exists = (
        SELECT COUNT(*) 
        FROM information_schema.STATISTICS 
        WHERE TABLE_SCHEMA = @dbname 
          AND TABLE_NAME = tbl_name 
          AND INDEX_NAME = idx_name
    );

        SET @sql = CONCAT('ALTER TABLE `', tbl_name, '` ADD INDEX `', idx_name, '` (', idx_definition, ')');
        PREPARE stmt FROM @sql;
        EXECUTE stmt;
        DEALLOCATE PREPARE stmt;
    END IF;
END$$

-- 1. Проверка дублей: если запрос вернёт строки — СТОП, пришлите вывод,
--    дам схлопывание. Иначе ALTER упадёт и индекс не создастся.
SELECT dict_name, COUNT(*) AS cnt
FROM lift_dict_versions
GROUP BY dict_name
HAVING cnt > 1;

-- 2. UNIQUE обязателен: без него INSERT ... ON DUPLICATE KEY UPDATE
--    не сработает как upsert, и два параллельных запроса вставят две строки
--    с одним ключом — инкременты начнут теряться.
ALTER TABLE lift_dict_versions
    ADD UNIQUE KEY uniq_dict_name (dict_name);

--    с версией 0 увидели outdated.

    ('dicts',          0, UNIX_TIMESTAMP()),

-- Миграция 1.3 — ротация refresh-токенов и детект переиспользования (B3.4).
-- Применять ДО выкатки UserRepository / AuthController: код читает новые колонки.
--
-- Зачем family_id, а не device_fingerprint: WebGL-отпечаток не уникален —
-- одинаковые GPU и драйверы дают один хэш. Отзыв по нему задел бы соседние
-- браузеры того же пользователя. Семья = одна цепочка ротаций от одного логина.
--
-- ascii_bin для hex-колонок: сравнение побайтовое и вчетверо компактнее
-- utf8mb4. Тот же приём, что у lift_object.guid.

ALTER TABLE lift_refresh_tokens
    ADD COLUMN family_id CHAR(32) CHARACTER SET ascii COLLATE ascii_bin NULL AFTER token_hash,
    ADD COLUMN rotated_at DATETIME NULL AFTER last_used_at,
    ADD COLUMN replaced_by CHAR(64) CHARACTER SET ascii COLLATE ascii_bin NULL AFTER rotated_at;

-- Ротации до этой миграции не было: каждая существующая строка — самостоятельная
-- семья. LEFT(token_hash, 32) уникален, потому что уникален token_hash (uq_token_hash).
-- RANDOM_BYTES не используем: детерминированный бэкфилл проще повторить и сверить.
UPDATE lift_refresh_tokens
SET family_id = LEFT(token_hash, 32)
WHERE family_id IS NULL;

-- NOT NULL ставим только после бэкфилла: иначе ALTER на непустой таблице
-- подставит '' и все старые строки окажутся в одной «семье» — отзыв одной
-- семьи разлогинил бы всех пользователей разом.
ALTER TABLE lift_refresh_tokens
    MODIFY COLUMN family_id CHAR(32) CHARACTER SET ascii COLLATE ascii_bin NOT NULL;

-- Дубли: idx_refresh_expires повторяет idx_expires_at, idx_refresh_user — idx_user_id.
-- Каждый лишний индекс — лишняя запись в B-tree на каждый refresh, а ротация
-- делает refresh чаще, чем раньше.
ALTER TABLE lift_refresh_tokens
    DROP INDEX idx_refresh_expires,
    DROP INDEX idx_refresh_user;

-- idx_refresh_family — отзыв семьи одним DELETE по (user_id, family_id).
-- idx_refresh_rotated — cron вычищает ротированные строки старше 7 дней:
-- окно детекта переиспользования = 7 дней, дальше хранить нечего.
ALTER TABLE lift_refresh_tokens
    ADD INDEX idx_refresh_family (user_id, family_id),
    ADD INDEX idx_refresh_rotated (rotated_at);

-- Кандидаты: улицы со скобками. Имя только из скобок не трогаем.
DROP TEMPORARY TABLE IF EXISTS tmp_street_fix;
CREATE TEMPORARY TABLE tmp_street_fix AS
SELECT id, city_id, street_type,
       TRIM(REGEXP_REPLACE(REGEXP_REPLACE(street_name, '\\([^)]*\\)', ''), '[[:space:]]+', ' ')) AS clean_name,
       NULLIF(TRIM(REGEXP_REPLACE(street_name, '^[^(]*\\(([^)]*)\\).*$', '$1')), '') AS note_new
FROM lift_street
WHERE street_name LIKE '%(%';
DELETE FROM tmp_street_fix WHERE clean_name = '';

ALTER TABLE tmp_street_fix ADD COLUMN new_key varchar(190) NULL;
UPDATE tmp_street_fix
SET new_key = CONCAT(COALESCE(street_type, ''), '|', LOWER(REPLACE(REPLACE(clean_name, ' ', ''), '.', '')));

-- Выживший id на каждый (city_id, new_key): минимум среди чистых и кандидатов.
DROP TEMPORARY TABLE IF EXISTS tmp_street_keep;
CREATE TEMPORARY TABLE tmp_street_keep AS
SELECT city_id, new_key, MIN(id) AS keep_id
FROM (
    SELECT id, city_id, new_key FROM tmp_street_fix
    UNION ALL
    SELECT s.id, s.city_id, s.street_key
    FROM lift_street s
    JOIN tmp_street_fix f ON s.city_id = f.city_id AND s.street_key = f.new_key
) u
GROUP BY city_id, new_key;

DROP TEMPORARY TABLE IF EXISTS tmp_street_map;
CREATE TEMPORARY TABLE tmp_street_map AS
SELECT f.id AS old_id, k.keep_id
FROM tmp_street_fix f
JOIN tmp_street_keep k ON k.city_id = f.city_id AND k.new_key = f.new_key;

-- Перепривязка домов и заявок на выжившую улицу.
UPDATE lift_home h
JOIN tmp_street_map m ON h.street_id = m.old_id AND m.old_id <> m.keep_id
SET h.street_id = m.keep_id;

UPDATE lift_calls c
JOIN tmp_street_map m ON c.address_street = m.old_id AND m.old_id <> m.keep_id
SET c.address_street = m.keep_id;

-- Чистое имя/ключ/примечание — выжившим кандидатам.
UPDATE lift_street s
JOIN tmp_street_fix f ON s.id = f.id
JOIN tmp_street_map m ON m.old_id = f.id AND m.keep_id = f.id
SET s.street_name = f.clean_name,
    s.street_key  = f.new_key,
    s.street_note = f.note_new;

-- Примечание в уже чистый выживший, если у него пусто.
UPDATE lift_street s
JOIN tmp_street_map m ON s.id = m.keep_id AND m.old_id <> m.keep_id
JOIN tmp_street_fix f ON f.id = m.old_id
SET s.street_note = COALESCE(NULLIF(s.street_note, ''), f.note_new);

DELETE s FROM lift_street s
