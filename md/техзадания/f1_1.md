## 1.16 — поколение в `refreshSilently`

````js
async refreshSilently(globalTicketsArrayRef) {
    if (this._isPolling) return;
    const myMountId = this._mountId;
    const isStale = () => this._mountId !== myMountId;
    this._isPolling = true;
    try {
        const fresh = await this._loadDataFromServer(true);
        if (!fresh || isStale()) return;

        let offlineTickets = [];
        if (this.dataType === "open") {
            const outbox = await offlineDB.getAllPending();
            if (isStale()) return;
            offlineTickets = outbox
                .filter((o) => o.command_type === "CreateTicket" && o.status === "pending")
                .map((o) => {
                    const createdAt = Number(o.created_at);
                    const dateInSeconds = createdAt > 9999999999 ? Math.floor(createdAt / 1000) : createdAt;
                    return {
                        ...o.payload,
                        id: o.offline_uuid || o.command_uuid,
                        type: "open",
                        isOffline: true,
                        address: o.payload.fullAddress || o.payload.address,
                        date: dateInSeconds,
                    };
                });
        }

        if (!Array.isArray(globalTicketsArrayRef)) return;

        globalTicketsArrayRef.splice(0, globalTicketsArrayRef.length, ...offlineTickets, ...fresh);
        await offlineDB.saveTickets(this.dataType, fresh);
        if (isStale()) return;

        const dataTable = document.getElementById("dataTable");
        if (dataTable) {
            this._renderTableFull(dataTable, globalTicketsArrayRef);
            if (this.currentSortColumnId !== -1) this._reApplyCurrentSort();
        }
    } catch (e) {
        console.error(`[TicketsController] refreshSilently failed:`, e);
    } finally {
        this._isPolling = false;
    }
}
````

## 1.6 — `outbox-queue.js`: `mergeIntoOfflineCreate` в одной `readwrite`-транзакции + без `repair_time`

```js
async function mergeIntoOfflineCreate(offlineUuid, updateFn) {
  const db = await offlineDB.dbPromise;
  const tx = db.transaction("outbox", "readwrite");
  const store = tx.objectStore("outbox");
  const index = store.index("offline_uuid");

  const cmds = await index.getAll(IDBKeyRange.only(offlineUuid));
  const cmd = cmds.find((c) => c.command_type === "CreateTicket");
  if (!cmd) {
    await tx.done;
    return null;
  }

  const updatedPayload = { ...cmd.payload };
  updateFn(updatedPayload);
  await store.put({ ...cmd, payload: updatedPayload });
  await tx.done;
  return cmd;
}
```

## 1.22 — `normalizeTicketFields`

Явно зафиксировать семантику 0:

```js
export function normalizeTicketFields(ticket) {
  if (!ticket) return ticket;

  const toIdOrNull = (v) => {
    if (v == null || v === "") return null;
    const n = Number(v);
    return Number.isFinite(n) && n !== 0 ? n : null;
  };
  const toStatus = (v) => {
    const n = Number(v);
    return Number.isFinite(n) ? n : 0;
  };

  ticket.staff_id = toIdOrNull(ticket.staff_id);
  ticket.staff_id2 = toIdOrNull(ticket.staff_id2);
  ticket.staff_id3 = toIdOrNull(ticket.staff_id3);

  ticket.notify_status = toStatus(ticket.notify_status);
  ticket.notify_status2 = toStatus(ticket.notify_status2);
  ticket.notify_status3 = toStatus(ticket.notify_status3);

  ticket.repair_time = toIdOrNull(ticket.repair_time);

  return ticket;
}
```

`toIdOrNull` для `staff_id` и `repair_time`: 0 = «не назначен» / «нет срока». `toStatus` для `notify_status`: 0 — валидное «не уведомлён».
