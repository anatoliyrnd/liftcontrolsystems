
**Одно уточнение решения по МСК (фиксирую явно).** В таблице даты рендерим `formatTs` (локальная зона) — как журналы: `body-render` форматирует колонки `dataType:"date"` сам, и править его нельзя (общий с журналами). МСК остаётся там, где он действительно нужен: границы периода (`mskDayStartTs/mskDayEndTs`), строка «Период МСК: …» под фильтрами (`formatTsMsk`) и CSV на сервере. Для пользователей в МСК это тождественно; расхождение UI≠CSV исчезает по построению, а не по соглашению. Подсветка просрочки `repair_time` при этом сохраняется (число, а не строка).

```js
function ymdMinusDays(days) {
  const d = new Date(Date.now() + 3 * 3600 * 1000 - days * 86400 * 1000);
  const pad = (n) => String(n).padStart(2, "0");
  return `${d.getUTCFullYear()}-${pad(d.getUTCMonth() + 1)}-${pad(d.getUTCDate())}`;
}

export function defaultReportState() {
  return {
    fromYmd: ymdMinusDays(REPORTS.DEFAULT_PERIOD_DAYS - 1),
    toYmd: mskTodayYmd(),
    dateType: "created",

    cityOnly: false,
    departmentIds: null,

    staffId: null,
    detailsText: "",
    solutionText: "",
    sort: REPORTS.SORT_DEFAULT,
    tab: REPORTS.TAB_DEFAULT,
    page: 1,
  };
}

export function loadReportState() {
  const saved = getReportFilters();
  const base = defaultReportState();
  if (!saved || typeof saved !== "object" || Array.isArray(saved)) return base;
```

```js
/**
 * Сводка отчёта. Чистый рендер: числа и подписи приходят с сервера,
 * DOM собирается createElement + textContent.
 *
 * NULL-корзины («Без отдела», «Без адреса»…) сервер отдаёт готовыми
 * подписями с id:null — здесь они ничем не отличаются от обычных
 * строк. Молча потерять их значило бы показать диспетчеру меньшее
 * число заявок, чем он видит в таблице.
 */
function el(tag, attrs = {}, text = null) {
  const node = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs)) {
    if (k === "class") node.className = v;
    else if (k === "style") node.style.cssText = v;
  }
  if (text != null) node.textContent = text;
  return node;
}

function fmtDuration(sec) {
  if (sec == null || !Number.isFinite(Number(sec))) return "—";
  const total = Math.round(Number(sec));
  const h = Math.floor(total / 3600);
  const m = Math.floor((total % 3600) / 60);
  return h > 0 ? `${h} ч ${m} мин` : `${m} мин`;
}

function card(title, value) {
  const box = el("div", {
    class: "rep-card",
    style:
      "background:var(--bg-secondary);border:2px solid var(--border-color);border-radius:10px;padding:10px 14px;min-width:130px;",
  });
  box.appendChild(el("div", {
    style: "font-size:11px;font-weight:700;color:var(--text-secondary);text-transform:uppercase;",
  }, title));
  box.appendChild(el("div", {
    style: "font-size:20px;font-weight:700;color:var(--text-primary);",
  }, String(value)));
  return box;
}

function groupBlock(title, rows) {
  if (!Array.isArray(rows) || rows.length === 0) return null;
  const box = el("div", { class: "rep-group", style: "min-width:200px;" });
  box.appendChild(el("div", {
    style: "font-size:11px;font-weight:700;color:var(--text-secondary);text-transform:uppercase;margin-bottom:4px;",
  }, title));
  for (const row of rows) {
    const line = el("div", {
      style: "display:flex;justify-content:space-between;gap:10px;font-size:12px;color:var(--text-primary);",
    });
    line.appendChild(el("span", {
      style: "overflow:hidden;text-overflow:ellipsis;white-space:nowrap;",
    }, String(row.name ?? "—")));
    line.appendChild(el("span", { style: "font-weight:600;" }, String(row.count ?? 0)));
    box.appendChild(line);
  }
  return box;
}

export function renderStats(host, stats) {
  host.innerHTML = "";
  if (!stats) {
    host.appendChild(el("div", {
      class: "rep-stats-empty",
      style: "font-size:13px;color:var(--text-secondary);margin:8px 0;",
    }, "Нет данных за выбранный период."));
    return;
  }
  const cards = el("div", {
    style: "display:flex;gap:10px;flex-wrap:wrap;margin:10px 0;",
  });
  cards.appendChild(card("Всего", stats.total ?? 0));
  cards.appendChild(card("Открыто", stats.open ?? 0));
  cards.appendChild(card("Закрыто", stats.closed ?? 0));
  cards.appendChild(card("Среднее время закрытия", fmtDuration(stats.avg_close_time_seconds)));
  const repair = stats.avg_repair_time || {};
  cards.appendChild(card(
    "Срок ремонта (сервер/диспетчер)",
    `${repair.default ?? 0} / ${repair.custom ?? 0}`,
  ));
  host.appendChild(cards);

  const grid = el("div", {
    class: "rep-stats-grid",
    style: "display:flex;gap:16px;flex-wrap:wrap;margin-bottom:10px;",
  });
  const blocks = [
    ["По отделам", stats.by_department],
    ["По городам", stats.by_city],
    ["По группам", stats.by_group],
    ["По типам запроса", stats.by_request],
  ];
  for (const [title, rows] of blocks) {
    const b = groupBlock(title, rows);
    if (b) grid.appendChild(b);
  }
  const levels = Array.isArray(stats.by_alert_level)
    ? stats.by_alert_level.map((r) => ({ name: `Уровень ${r.level}`, count: r.count }))
    : [];
  const lb = groupBlock("По уровням тревоги", levels);
  if (lb) grid.appendChild(lb);
  if (grid.childElementCount > 0) host.appendChild(grid);
}
```

```js
import { REPORTS } from "../../config.js";
import { setScope, setReportColumns } from "../../table/settings.js";
import { rebuildTableHeaders } from "../../table/header-render.js";
import { renderTableBody } from "../../table/body-render.js";

/**
 * Таблица отчёта переиспользует header-render/body-render: липкие
 * колонки, ширины per-user и подсветка тревоги достаются бесплатно.
 *
 * Своё здесь только одно — гарантия источника колонок: setScope +
 * setReportColumns перед КАЖДЫМ рендером, по образцу _renderTableFull
 * журналов. applyUserSettings сбрасывает scope на "open" и обнуляет
 * reportSource при смене пользователя, поэтому точка гарантии одна.
 *
 * Колонки sortable:false целиком ⇒ header-render не вешает клики,
 * sorting.js не вызывается: сортировка отчёта серверная, направление
 * фиксировано белым списком REPORTS.SORTS.
 */
export function createTableView(table, router) {
  let rowClickAbort = null;

  function ensureColumns() {
    setScope("report");
    setReportColumns(REPORTS.COLUMNS);
  }

  function rebuildHeaders() {
    ensureColumns();
    const old = table.querySelector("thead");
    if (old) old.remove();
    // onSortCallback не вызывается никогда (sortable:false), но
    // сигнатура обязательна — передаём noop осознанно.
    rebuildTableHeaders(table, () => {}, () => {});
  }

  function renderRows(items) {
    ensureColumns();
    renderTableBody(table, Array.isArray(items) ? items : []);
  }

  function bindRowClick() {
    if (rowClickAbort) rowClickAbort.abort();
    rowClickAbort = new AbortController();
    const tbody = table.querySelector("tbody");
    if (!tbody) return;
    tbody.addEventListener(
      "click",
      (e) => {
        if (e.target.classList.contains("resizer")) return;
        const row = e.target.closest("tr");
        if (!row) return;
        const idCell = row.querySelector('td[data-col-id="id"]');
        const id = idCell ? idCell.textContent.trim() : "";
        // Только серверные номера: роутер валидирует param строго.
        if (!/^\d+$/.test(id)) return;
        router.navigate("#/ticket/" + encodeURIComponent(id));
      },
      { signal: rowClickAbort.signal },
    );
  }

  function renderPagination(container, pagination, onPage) {
    container.innerHTML = "";
    if (!pagination || !pagination.total_items) return;
    const page = pagination.current_page;
    const pages = Math.max(1, pagination.total_pages);
    const prev = document.createElement("button");
    prev.type = "button";
    prev.className = "theme-toggle-btn";
    prev.style.cssText = "border-radius:6px;width:auto;padding:0 10px;font-size:13px;";
    prev.textContent = "‹";
    prev.disabled = page <= 1;
    prev.addEventListener("click", () => onPage(page - 1));
    const next = document.createElement("button");
    next.type = "button";
    next.className = "theme-toggle-btn";
    next.style.cssText = "border-radius:6px;width:auto;padding:0 10px;font-size:13px;";
    next.textContent = "›";
    next.disabled = page >= pages;
    next.addEventListener("click", () => onPage(page + 1));
    const info = document.createElement("span");
    info.textContent = `Стр. ${page} из ${pages} (всего ${pagination.total_items})`;
    container.appendChild(prev);
    container.appendChild(info);
    container.appendChild(next);
  }

  function destroy() {
    if (rowClickAbort) rowClickAbort.abort();
    rowClickAbort = null;
  }

  return { ensureColumns, rebuildHeaders, renderRows, bindRowClick, renderPagination, destroy };
}
```
